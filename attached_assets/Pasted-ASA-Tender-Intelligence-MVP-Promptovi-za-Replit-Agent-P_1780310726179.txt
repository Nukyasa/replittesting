export const ejnApi = {
  getAnnouncements: async (params: Record<string, string> = {}) => {
    const defaults = { '$format': 'json', '$orderby': 'PublicationDate desc' };
    const q = new URLSearchParams({ ...defaults, ...params }).toString();
    const res = await fetch(`${BASE}/Announcements?${q}`);
    if (!res.ok) throw new Error(`EJN error: ${res.status}`);
    return res.json();
  },
  getAnnouncementById: async (id: number) => {
    const res = await fetch(`${BASE}/Announcements(${id})?$format=json`);
    if (!res.ok) throw new Error(`EJN error: ${res.status}`);
    return res.json();
  },
  getAuthorities: async (params = {}) => {
    const q = new URLSearchParams({ '$format': 'json', ...params }).toString();
    const res = await fetch(`${BASE}/ContractingAuthorities?${q}`);
    return res.json();
  },
  getStats: async () => {
    const res = await fetch(`${BASE}/Statistics?$format=json`);
    return res.json();
  }
};

```typescript
const buildFilter = (f: Filters): string => {
  const parts: string[] = [];
  if (f.search) parts.push(`contains(tolower(Subject),tolower('${f.search}'))`);
  if (f.status) parts.push(`StatusId eq ${f.status}`);
  if (f.dateFrom) parts.push(`PublicationDate ge ${f.dateFrom}T00:00:00Z`);
  if (f.dateTo) parts.push(`PublicationDate le ${f.dateTo}T23:59:59Z`);
  if (f.minValue) parts.push(`EstimatedValue ge ${f.minValue}`);
  if (f.maxValue) parts.push(`EstimatedValue le ${f.maxValue}`);
  return parts.join(' and ');
};
```

```typescript
const CountdownTimer = ({ deadline }: { deadline: string }) => {
  const [timeLeft, setTimeLeft] = useState('');
  useEffect(() => {
    const interval = setInterval(() => {
      const diff = new Date(deadline).getTime() - Date.now();
      if (diff <= 0) { setTimeLeft('Isteklo'); return; }
      const d = Math.floor(diff / 86400000);
      const h = Math.floor((diff % 86400000) / 3600000);
      const m = Math.floor((diff % 3600000) / 60000);
      setTimeLeft(`${d}d ${h}h ${m}m`);
    }, 1000);
    return () => clearInterval(interval);
  }, [deadline]);
  return <span className={timeLeft === 'Isteklo' ? 'text-red-600' : 'text-orange-600'}>{timeLeft}</span>;
};
```

1. STORAGE (koristiti localStorage):
```typescript
const WATCHLIST_KEY = 'asa_tender_watchlist_v2';

interface WatchedTender {
  Id: number;
  Subject: string;
  ContractingAuthorityName: string;
  DeadlineDate: string;
  EstimatedValue: number;
  CurrencyCode: string;
  StatusId: number;
  StatusName: string;
  addedAt: string;
  lastChecked?: string;
  hasUpdate?: boolean;
  updateType?: 'status_changed' | 'deadline_changed' | 'value_changed';
}

export const watchlistStorage = {
  getAll: (): WatchedTender[] => JSON.parse(localStorage.getItem(WATCHLIST_KEY) || '[]'),
  add: (tender: EjnAnnouncement) => {
    const list = watchlistStorage.getAll();
    if (!list.find(t => t.Id === tender.Id)) {
      list.push({ ...tender, addedAt: new Date().toISOString() });
      localStorage.setItem(WATCHLIST_KEY, JSON.stringify(list));
    }
  },
  remove: (id: number) => {
    const list = watchlistStorage.getAll().filter(t => t.Id !== id);
    localStorage.setItem(WATCHLIST_KEY, JSON.stringify(list));
  },
  update: (id: number, changes: Partial<WatchedTender>) => {
    const list = watchlistStorage.getAll().map(t => t.Id === id ? {...t, ...changes} : t);
    localStorage.setItem(WATCHLIST_KEY, JSON.stringify(list));
  },
  isWatched: (id: number): boolean => watchlistStorage.getAll().some(t => t.Id === id)
};
```

2. PROVJERA AŽURIRANJA:
```typescript
const checkUpdates = async () => {
  setChecking(true);
  const list = watchlistStorage.getAll();
  const results = await Promise.allSettled(
    list.map(async (saved) => {
      const res = await fetch(`/api/ejn/Announcements(${saved.Id})?$format=json`);
      const current = await res.json();
      const changes: string[] = [];
      if (current.StatusId !== saved.StatusId) changes.push('Status promijenjen');
      if (current.DeadlineDate !== saved.DeadlineDate) changes.push('Rok promijenjen');
      if (current.EstimatedValue !== saved.EstimatedValue) changes.push('Vrijednost promijenjena');
      if (changes.length > 0) {
        watchlistStorage.update(saved.Id, { 
          ...current, 
          hasUpdate: true, 
          updateType: 'status_changed',
          lastChecked: new Date().toISOString()
        });
      }
      return { id: saved.Id, changes };
    })
  );
  setTenders(watchlistStorage.getAll());
  setChecking(false);
  const total = results.filter(r => r.status === 'fulfilled' && r.value.changes.length > 0).length;
  if (total > 0) toast(`${total} tender(a) ima promjene!`);
};
```

```typescript
import jsPDF from 'jspdf';

export interface TenderForExport {
  Id: number;
  Subject?: string;
  ContractingAuthorityName?: string;
  PublicationDate?: string;
  DeadlineDate?: string;
  EstimatedValue?: number;
  CurrencyCode?: string;
  StatusName?: string;
  ProcurementType?: string;
  CpvCode?: string;
  CpvName?: string;
  HasEAuction?: boolean;
  AwardCriteria?: string;
}

const formatBAM = (value?: number, currency?: string): string => {
  if (!value) return 'N/A';
  return `${new Intl.NumberFormat('de-DE').format(value)} ${currency || 'BAM'}`;
};

const formatDate = (dateStr?: string): string => {
  if (!dateStr) return 'N/A';
  return new Date(dateStr).toLocaleDateString('bs-BA');
};

// JEDAN TENDER - detaljan PDF
export const exportSingleTenderPDF = (tender: TenderForExport): void => {
  const doc = new jsPDF('p', 'mm', 'a4');
  const W = 210, M = 20;
  let y = 0;

  // Header plavi banner
  doc.setFillColor(27, 58, 107);
  doc.rect(0, 0, W, 40, 'F');
  doc.setTextColor(255, 255, 255);
  doc.setFontSize(18); doc.setFont('helvetica', 'bold');
  doc.text('ASA Tender Intelligence', M, 16);
  doc.setFontSize(10); doc.setFont('helvetica', 'normal');
  doc.text('Detalji tendera', M, 26);
  doc.text(new Date().toLocaleDateString('bs-BA'), W-M, 26, { align: 'right' });
  
  // Zlatna linija
  doc.setFillColor(232, 160, 32);
  doc.rect(0, 40, W, 2, 'F');
  y = 52;

  // Naziv
  doc.setTextColor(27, 58, 107);
  doc.setFontSize(14); doc.setFont('helvetica', 'bold');
  const titleLines = doc.splitTextToSize(tender.Subject || 'Bez naziva', W - 2*M);
  doc.text(titleLines, M, y);
  y += titleLines.length * 7 + 8;

  // Info tabela
  doc.setFontSize(10);
  const rows = [
    ['Ugovorni organ', tender.ContractingAuthorityName || 'N/A'],
    ['Datum objave', formatDate(tender.PublicationDate)],
    ['Rok za ponude', formatDate(tender.DeadlineDate)],
    ['Procij. vrijednost', formatBAM(tender.EstimatedValue, tender.CurrencyCode)],
    ['Tip nabavke', tender.ProcurementType || 'N/A'],
    ['CPV kod', `${tender.CpvCode || ''} ${tender.CpvName || ''}`],
    ['Kriterij dodjele', tender.AwardCriteria || 'N/A'],
    ['E-aukcija', tender.HasEAuction ? 'Da' : 'Ne'],
    ['Status', tender.StatusName || 'N/A'],
    ['EJN ID', String(tender.Id)],
  ];

  rows.forEach(([label, value], i) => {
    const bg = i % 2 === 0 ? [245, 247, 250] : [255, 255, 255];
    doc.setFillColor(...bg as [number,number,number]);
    doc.rect(M, y-4, W-2*M, 9, 'F');
    doc.setTextColor(100);
    doc.setFont('helvetica', 'bold');
    doc.text(label + ':', M+2, y+2);
    doc.setTextColor(0);
    doc.setFont('helvetica', 'normal');
    const valLines = doc.splitTextToSize(String(value), W - M - 80);
    doc.text(valLines, M+70, y+2);
    y += Math.max(valLines.length * 5, 9) + 1;
  });

  // EJN Link
  y += 5;
  doc.setTextColor(27, 58, 107);
  doc.setFontSize(9);
  doc.text('Link na EJN portal:', M, y);
  const link = `https://next.ejn.gov.ba/bs-latn-ba/procurements/announcement/${tender.Id}`;
  doc.textWithLink(link, M, y+6, { url: link });

  // Footer
  doc.setFontSize(8); doc.setTextColor(150);
  doc.line(M, 285, W-M, 285);
  doc.text(`Generirano: ${new Date().toLocaleString('bs-BA')} | ASA Tender Intelligence`, M, 290);

  doc.save(`tender-${tender.Id}.pdf`);
};

// BATCH EXPORT — više tendera u jedan PDF
export const exportBatchTendersPDF = (tenders: TenderForExport[], title = 'Izvještaj o tenderima'): void => {
  const doc = new jsPDF('l', 'mm', 'a4'); // landscape za tabelu
  const W = 297, M = 15;
  
  // Header
  doc.setFillColor(27, 58, 107);
  doc.rect(0, 0, W, 30, 'F');
  doc.setTextColor(255, 255, 255);
  doc.setFontSize(16); doc.setFont('helvetica', 'bold');
  doc.text('ASA Tender Intelligence', M, 12);
  doc.setFontSize(10); doc.setFont('helvetica', 'normal');
  doc.text(title, M, 22);
  doc.text(`${tenders.length} tendera | ${new Date().toLocaleDateString('bs-BA')}`, W-M, 22, {align:'right'});
  
  // Tabela header
  let y = 40;
  const cols = [10, 90, 55, 28, 30, 35, 25]; // widths
  const headers = ['ID', 'Naziv', 'Ugovorni organ', 'Datum', 'Rok', 'Vrijednost', 'Status'];
  
  doc.setFillColor(232, 160, 32);
  doc.rect(M, y, W-2*M, 8, 'F');
  doc.setTextColor(27, 58, 107);
  doc.setFontSize(8); doc.setFont('helvetica', 'bold');
  let x = M + 2;
  headers.forEach((h, i) => { doc.text(h, x, y+5); x += cols[i]; });
  y += 10;

  // Redovi
  tenders.forEach((t, idx) => {
    if (y > 185) { doc.addPage(); y = 20; }
    const bg = idx % 2 === 0 ? [245, 247, 250] : [255, 255, 255];
    doc.setFillColor(...bg as [number,number,number]);
    doc.rect(M, y, W-2*M, 7, 'F');
    doc.setTextColor(0); doc.setFont('helvetica', 'normal'); doc.setFontSize(7);
    const cells = [
      String(t.Id),
      (t.Subject || '').slice(0, 45),
      (t.ContractingAuthorityName || '').slice(0, 30),
      formatDate(t.PublicationDate),
      formatDate(t.DeadlineDate),
      formatBAM(t.EstimatedValue, t.CurrencyCode),
      t.StatusName || ''
    ];
    x = M + 2;
    cells.forEach((c, i) => { doc.text(c, x, y+4.5); x += cols[i]; });
    y += 8;
  });

  // Footer
  const pages = doc.getNumberOfPages();
  for (let i = 1; i <= pages; i++) {
    doc.setPage(i);
    doc.setFontSize(7); doc.setTextColor(150);
    doc.text(`Stranica ${i}/${pages}`, W-M, 205, { align: 'right' });
  }

  doc.save(`asa-tender-izvjestaj-${new Date().toISOString().split('T')[0]}.pdf`);
};
```

```typescript
const SETTINGS_KEY = 'asa_tender_settings';

interface AppSettings {
  theme: 'light' | 'dark' | 'system';
  language: 'bs' | 'en';
  anthropicApiKey: string;
  defaultStatus: '' | '1' | '2';
  defaultPageSize: 20 | 50 | 100;
  notificationsEnabled: boolean;
}

const DEFAULT_SETTINGS: AppSettings = {
  theme: 'light',
  language: 'bs',
  anthropicApiKey: '',
  defaultStatus: '1',
  defaultPageSize: 20,
  notificationsEnabled: true,
};

export const settingsStorage = {
  get: (): AppSettings => ({ ...DEFAULT_SETTINGS, ...JSON.parse(localStorage.getItem(SETTINGS_KEY) || '{}') }),
  set: (s: Partial<AppSettings>) => localStorage.setItem(SETTINGS_KEY, JSON.stringify({ ...settingsStorage.get(), ...s })),
  reset: () => localStorage.removeItem(SETTINGS_KEY),
};
```
