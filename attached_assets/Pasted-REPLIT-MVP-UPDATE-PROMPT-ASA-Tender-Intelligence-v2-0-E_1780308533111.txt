1.1 EJN Open Data API (PRIMARNI IZVOR)
Koristi OData REST API sa https://open.ejn.gov.ba — bez auth, direktni fetch.
javascript// server/services/ejnService.js

const EJN_BASE = 'https://open.ejn.gov.ba';

export async function fetchTenders({ top = 20, skip = 0, search = '', statusId = null } = {}) {
  const params = new URLSearchParams({
    '$top': top,
    '$skip': skip,
    '$orderby': 'PublicationDate desc',
    '$format': 'json',
    '$count': 'true'
  });

  const filters = [];
  if (search) filters.push(`contains(Subject,'${search}')`);
  if (statusId) filters.push(`StatusId eq ${statusId}`);
  if (filters.length) params.set('$filter', filters.join(' and '));

  const res = await fetch(`${EJN_BASE}/Announcements?${params}`);
  if (!res.ok) throw new Error('EJN API error: ' + res.status);
  const data = await res.json();
  return { tenders: data.value, total: data['@odata.count'] };
}

export async function fetchTenderById(id) {
  const res = await fetch(`${EJN_BASE}/Announcements(${id})?$format=json&$expand=Documents,Lots`);
  if (!res.ok) throw new Error('EJN tender not found');
  return res.json();
}
1.2 EJN Polja koja treba mapirati
Svaki Announcement objekat ima:

Id — jedinstven ID
Subject — naziv tendera
PublicationDate — datum objave
DeadlineDate — rok za prijem ponuda
QuestionsDeadline — rok za postavljanje pitanja
EstimatedValue — procijenjena vrijednost (broj)
CurrencyCode — valuta (BAM, EUR)
ContractingAuthorityName — naziv ugovornog organa
ProcurementType / ProcurementTypeName — vrsta postupka
StatusId / StatusName — status (1=Aktivan, 2=Zatvoren, 3=Poništen)
CpvCode / CpvName — CPV kategorija
HasEAuction — boolean, ima li e-aukciju
AnnouncementUrl — link na EJN portal
AwardCriteria — kriterij dodjele (najniža cijena / ekonomski najpovoljnija)
AwardCriteriaDetails — detalji kriterija (težine, ponderacija)
TenderPreparationCost — troškovi pripreme ponude
RequiredGuaranteeAmount — iznos garancije
RequiredGuaranteeType — vrsta garancije
Documents (expand) — lista dokumenata (ANNEX I, tenderske dokumentacije)
Lots (expand) — lotovi / dijelovi tendera

model TenderDocument {
  id          String  @id @default(cuid())
  tenderId    String
  tender      Tender  @relation(fields: [tenderId], references: [id])
  name        String
  type        String  // ANNEX_I, TENDER_DOC, etc.

  ejnDocId    Int?

model TenderChange {
  id          String   @id @default(cuid())

  field       String
  oldValue    String?
  newValue    String?
  changedAt   DateTime @default(now())
  notified    Boolean  @default(false)
}

model TenderNotification {
  id          String   @id @default(cuid())

  userId      String
  type        String   // DEADLINE_REMINDER, CHANGE_DETECTED, NEW_TENDER
  message     String
  read        Boolean  @default(false)
  createdAt   DateTime @default(now())
}

router.get('/:id', async (req, res) => {

  let tender = await prisma.tender.findUnique({

2.2 Cron sync — server/jobs/syncTenders.js
javascriptimport cron from 'node-cron';
import { fetchTenders, fetchTenderById } from '../services/ejnService.js';
import { prisma } from '../db.js';

// Svaki sat: provjeri za izmjene na praćenim tenderima
cron.schedule('0 * * * *', async () => {
  console.log('[SYNC] Provjera izmjena na tenderima...');
  
  const trackedTenders = await prisma.tender.findMany({
    where: { statusId: 1 }, // samo aktivni
    select: { ejnId: true, deadlineDate: true, subject: true }
  });
  
  for (const t of trackedTenders) {
    try {
      const live = await fetchTenderById(t.ejnId);
      await detectAndSaveChanges(t, live);
    } catch (e) {
      console.error(`[SYNC] Greška za tender ${t.ejnId}:`, e.message);
    }
  }
});

// Svaki dan u 8:00 — check rokovi
cron.schedule('0 8 * * *', async () => {
  const soon = new Date();
  soon.setDate(soon.getDate() + 3);
  
  const deadlineSoon = await prisma.tender.findMany({
    where: {
      statusId: 1,
      deadlineDate: { gte: new Date(), lte: soon }
    }
  });
  
  for (const t of deadlineSoon) {
    await createDeadlineNotification(t);
  }
});

async function detectAndSaveChanges(old, live) {
  const fields = ['deadlineDate', 'questionsDeadline', 'estimatedValue', 'statusId'];
  const changes = [];
  
  for (const field of fields) {
    if (String(old[field]) !== String(live[field])) {
      changes.push({ field, oldValue: String(old[field]), newValue: String(live[field]) });
    }
  }
  
  if (changes.length > 0) {
    await prisma.tenderChange.createMany({
      data: changes.map(c => ({ ...c, tenderId: old.id }))
    });
    
    await prisma.tenderNotification.create({
      data: {
        tenderId: old.id,
        userId: 'all', // ili specifičan user
        type: 'CHANGE_DETECTED',
        message: `Izmjene na tenderu: ${old.subject} — ${changes.map(c => c.field).join(', ')}`
      }
    });
  }
}

javascript// server/scrapers/ejnAuthScraper.js
import { chromium } from 'playwright';

export async function scrapeFullTenderDetails(tenderId) {
  const browser = await chromium.launch({ headless: true });
  const page = await browser.newPage();
  
  // Login
  await page.goto('https://www.ejn.gov.ba/');
  await page.click('[href*="login"]');
  await page.fill('input[name="username"]', 'almir.zeljkovic');
  await page.fill('input[name="password"]', 'Start.2024');
  await page.click('button[type="submit"]');
  await page.waitForNavigation();
  
  // Preuzmi detalje tendera
  await page.goto(`https://www.ejn.gov.ba/tender/${tenderId}`);
  
  const details = await page.evaluate(() => ({
    conditions: document.querySelector('.participation-conditions')?.innerText,
    documents: Array.from(document.querySelectorAll('.document-list a')).map(a => ({
      name: a.innerText,
      url: a.href
    }))
  }));
  
  await browser.close();
  return details;
}
